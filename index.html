type MityaState = {
	position: 'hiding' | 'running' | 'impaled'
	seesChainsaw: boolean
	panicLevel: number
};

type MityaAction =
	| { type: 'SEE_CHAINSAW' }
	| { type: 'BEE_STING' }
	| { type: 'RUN_AWAY' }
	| { type: 'LOOK_WHERE_RUNNING' }
	| { type: 'HIT_OBSTACLE'; obstacle: string }

function mityaReducer(state: MityaState, action: MityaAction): MityaState {
	switch (action.type) {
		case 'SEE_CHAINSAW':
			return { ...state, seesChainsaw: true, panicLevel: 10 }

		case 'RUN_AWAY':
			return { ...state, position: 'running' }

		case 'LOOK_WHERE_RUNNING':
			return { ...state, panicLevel: state.panicLevel - 3 }

		case 'HIT_OBSTACLE':
			if (state.position === 'running') {
				return { ...state, position: 'impaled' } // 🌿 ветка
			}

			return state

		default:
			return state
	}
}

}

Каждый менеджер мог спасти своего героя
  • Василиса, useState — действие до завершения async-проверки | один флаг-этап `phase` вместо гонки состояний
  • Митя, useReducer — переход без guard'а | guard в кейсе `RUN_AWAY`
  • Толя, Redux — устаревшие координаты при атаке | requestId в экшенах, отбрасывать поздние ответы
  • Михаил, Zustand — «отмена», не отменяющая таймер | `clearTimeout` рядом с переходом в `running`
  • Яша, MobX — каскад reactions без disposer'ов | сохранить disposer'ы, останавливать панику
  • Нина, Jotai — цикл мнения ⇄ угрозы | ациклический граф: мнение → угроза, без обратной петли
  • Чеслав, Recoil — мёртвая библиотека | миграция до хакатона («никак» — в этом шутка)
  • Потап, Effector — хрупкая сеть из 6 узлов | один derived-стор вместо цепочки sample
  • —, XState — «не серебряная пуля» | забытый `isAllergic` в контексте убивает и Алису
